aboutmonitor
description: returns info about resolution, available size and depth
syntax: aboutmonitor()
example:
aboutmonitor()
-- if exist an opened window, returns the following record:
{
    Resolution:{
        1024,
        768
    },
    AvailableSize:{
        1011,
        741
    },
    Colores:4.294967296E+9
}

displayinfo
description: makes a window and writes some text to it
syntax: displayinfo(windowtitle,message,windowbounds,timeout)

windowtitle:
string
message:string/html formatted
windowbounds:list of integer = {distance from left,distance from top,weight,heigth}
timeout:integer (seconds) -- 0 means "without timeout"
example:
displayinfo("Hei", "This is a message made for you, guy!", {50, 20, 300, 300}, 5)
-- on error, returns 0

displayconfirm
description: displays a java confirm dialog
syntax: displayconfirm(message)
message:
string
example:
displayconfirm("Hace mucho calor...")
-- returns true or false, depending on user's election

displayprompt
description: displays a java prompt dialog
syntax: displayprompt(message,defaultanswer)
message:
string
defaultanswer:string
example:
displayprompt("What's your name?","Bob Dylan")
-- returns the user's answer or "null", if he cancelled

loadpage
description: If 0, reload the current page (look at syntax). If -1, go to the last page in the browser's history; if 1, to the next page; blah, blah, blah.
syntax: loadpage(history)
history:
integer
example:
loadpage(-5)
-- go to page -5 in history's browser

location
description: go to the specified URL at the specified frame
syntax: location(frame,URL)
frame:
string, the name of the frame. If 0 (zero), opens the URL in the top of current page
URL:string, the url
example:
location("leftframe","http://www.apple.com/")
-- opens the url in frame "leftframe"

resizewin
description: change the size of the current page
syntax: resizewin to {width, heigth, centered}
width:
integer (pixels)
heigth:integer (pixels)
centered:boolean (true/false)
example:
resizewin to {200, 500, true}
-- on error, returns 0

movewin
description: move the front window to the specified coordinates
syntax: movewin to {x, y}
x:
integer (pixels), distance from left
y:integer (pixels), distance from top
example:
movewin to {200, 500}
-- on error, returns 0

getcolor
description: get the hexadecimal color of some element
syntax: getcolor(element)
element:
string, the element, wich can be "bg" (background), "fg" (foreground), "link" (link), "alink" (active link) & "vlink" (visited link)
example:
getcolor("fg")
-- returns "#FF0000"

setcolor
description: set the color (in hexadecimal notation, with or without "#") of some element
syntax: setcolor(element,color)
element:
string, the element, wich can be "bg" (background), "fg" (foreground), "link" (link), "alink" (active link) & "vlink" (visited link)
color:string, hexadecimal color
example:
setcolor("fg", "123456")
-- returns the applied color

bookmark
description: adds the current page to your favorites
syntax: bookmark()
example:
bookmark()
-- on error, returns 0
